kubectl endpoints

# check persistence of file on hostPath
kubectl get pods -n k8s-program
kubectl exec songs-ms-78b67c5844-4wskb -n k8s-program -- sh -c "echo 'This file persists' > /app/data/test.txt"
kubectl exec songs-ms-78b67c5844-4wskb -n k8s-program -- cat /app/data/test.txt
kubectl delete pod songs-ms-78b67c5844-4wskb -n k8s-program
kubectl get pods -n k8s-program
kubectl exec songs-ms-78b67c5844-lfb89 -n k8s-program -- cat /app/data/test.txt

kubectl apply --dry-run (best for K8s validation)

# Validate YAML syntax
kubectl apply --dry-run=client -f k8s/
